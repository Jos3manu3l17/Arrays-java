import java.util.Scanner;

public class Caso111 {

  public static void main(String[] args) {

    Scanner jm = new Scanner(System.in);

    // Caso 111. Sismología Rural: Mapeo de Intensidad de Micro-Sismos Superficiales Una red de sensores monitorea una zona de influencia volcánica dividida en una cuadrícula de 10x10 kilómetros. En cada celda de la matriz, se registra la magnitud máxima de los microsismos detectados en las últimas 24 horas. El aprendiz debe desarrollar un algoritmo que: 1. Identifique el "Epicentro de Actividad" (la celda con el valor más alto). 2. Calcule la magnitud promedio de la zona para determinar el nivel de alerta (Verde, Amarillo, Naranja). 3. Detectar patrones de propagación lineal que podrían indicar la activación de una falla geológica oculta, señalando las coordenadas exactas de mayor riesgo para la evacuación.

    System.out.println("\n===== Inicio del sistema de sismologia rural =====");

            double[][] sismos = new double[10][10]; 

        double suma = 0; 

        double max = Double.MIN_VALUE; 
        int filaMax = 0, colMax = 0;  

        for (int i = 0; i < 10; i++) {
            for (int j = 0; j < 10; j++) {
                System.out.print("Magnitud en [" + i + "][" + j + "]: ");
                sismos[i][j] = jm.nextDouble();

                suma += sismos[i][j];

                if (sismos[i][j] > max) {
                    max = sismos[i][j];
                    filaMax = i;
                    colMax = j;
                }
            }
        }

        double promedio = suma / 100;

        String alerta;

        if (promedio < 2) {
            alerta = "Verde";
        } else if (promedio < 4) {
            alerta = "Amarillo";
        } else {
            alerta = "Naranja";
        }

        System.out.println("\nEPICENTRO DE ACTIVIDAD:");
        System.out.println("Magnitud: " + max + " en coordenadas [" + filaMax + "][" + colMax + "]");

        System.out.println("\nPromedio de la zona: " + promedio);
        System.out.println("Nivel de alerta: " + alerta);

        System.out.println("\nPosibles patrones lineales detectados:");

        for (int i = 0; i < 10; i++) { 
            for (int j = 0; j < 9; j++) { 

                double diferencia = Math.abs(sismos[i][j] - sismos[i][j + 1]);

                if (diferencia < 0.5) {
                    System.out.println("Posible línea en [" + i + "][" + j + "] y [" + i + "][" + (j + 1) + "]");
                }
            }
        }

        jm.close(); 
  }
}
